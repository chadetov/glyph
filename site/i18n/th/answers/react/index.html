ใช้กับ React ได้ไหม: component, hook และ Context?

ได้ component คอมไพล์เป็น function component ของ React, JSX ถูกแปลงเป็น createElement, hook ของ React เองถูก import ด้วยชื่อและ destructure คู่ค่าของมันได้ และ state ของ reducer ได้การตรวจความครบถ้วนที่ TypeScript ธรรมดาไม่มี ตัวหน้าเว็บเอง (title, จุด mount, state ที่บันทึกไว้) เข้าถึงผ่าน std/dom และ std/storage ช่องว่างที่ควรรู้ก่อน: ตัวตรวจของ Glyph เองไม่ได้ให้ type กับสิ่งที่ hook คืนมา tsc จึงเป็นคนตรวจค่าเหล่านั้น

ทำไมเรื่องนี้สำคัญ

โค้ดที่เอเจนต์เขียนจำนวนมากเป็น React Glyph ถือว่ามันเป็นเป้าหมายหลัก ไม่ใช่ของที่แปะเพิ่ม: component เป็น keyword ของตัวเอง (grep "^component " จึงตรวจ UI ทั้งหมดได้) hook ของ React import จาก react และเรียกด้วยชื่อและ type จริง ตรวจโดย tsc เทียบกับ declaration ของ React เอง

ดูของจริง

useReducer ของ React จริง กับ reducer บน tagged union ทุก action ต้องถูกจัดการ และคอมไพเลอร์ตรวจให้:

counter.glyph กับ react@18 และ type ของมัน 
import react { useReducer }

type Action = Inc | Dec | Reset

fn reducer(n: number, a: Action) -> number {
  return match a {
    Inc => n + 1,
    Dec => n - 1,
    Reset => 0,
    // add an Action variant and this match is E0200
  }
}

pub component Counter() {
  let [count, dispatch] = useReducer(reducer, 0)
  return <button on_click={fn() { dispatch(Inc) }}>{count}</button>
}

เพิ่ม action Double แล้ว reducer จะคอมไพล์ไม่ผ่านด้วย E0200 ที่ระบุชื่อมัน ซึ่ง switch ของ TypeScript ธรรมดาจะปล่อยให้ออกไป คู่ค่าถูก destructure ด้วย let [count, dispatch] และ TypeScript ที่ emit ออกมาเพิ่ม satisfies readonly [unknown, unknown] tsc จึงพิสูจน์ได้ว่า hook คืนคู่ค่าและยังอนุมาน type ของแต่ละตัวได้ ตัวตรวจของ Glyph เองไม่ให้ type กับค่าที่มาจาก React argument ที่ผิดสำหรับ dispatch หรือ state setter จึงถูกรายงานโดย tsc ใน build ไม่ใช่โดย editor หรือ --no-tsc React Context ใช้ได้วันนี้ผ่าน createContext / useContext และชื่อ element แบบ member expression (<ThemeContext.Provider>) ใช้ได้โดยตรง ไม่ต้องหาทางเลี่ยง

การ spread prop ใน JSX ก็ใช้ได้: <input {...register("email")} /> ถูกแปลงเป็น object spread ใน props idiom ของ react-hook-form จึงไม่ต้องมี adapter type ที่ derive จากค่าของ zod ใช้ได้เต็มที่: type FormData = z.infer<typeof schema> ใช้ได้โดยตรง ไม่ต้องใช้ทางออกฉุกเฉิน สิ่งเดียวที่ต่าง: component รับ props record ตัวเดียว ไม่ใช่หลาย parameter ตามตำแหน่ง ดูการใช้ไลบรารีของคุณและหน้าการทดสอบความแข็งแรงว่าขอบเขตนั้นอยู่ตรงไหนพอดี

props ถูกตรวจก่อน tsc ไหม?

ใช่ Glyph เทียบแต่ละ element กับ props record ของ component เอง ค่าที่ผิด prop ที่สะกดผิด และ prop บังคับที่ขาดไป จึงเป็น error ของ Glyph ที่ระบุชื่อ prop คุณเห็นมันใน editor และผ่าน MCP ไม่ใช่แค่ใน build เต็ม และทำงานเหมือนกันไม่ว่า component จะถูก import ด้วยชื่อหรือผ่าน module ของมัน:

main.glyph โดยมี Card ประกาศไว้ใน ui.glyph 
import ui

pub component App() {
  return <div>
    <ui.Card title="Q3" />
    <ui.Card title="Q4" cout={2} />
  </div>
}

// E0211 argument type mismatch: expected `CardProps`, found `record`: missing field `count`
// E0210 type `CardProps` has no field `cout`; did you mean `count`?
// E0211 argument type mismatch: expected `CardProps`, found `record`: missing field `count`

--json ส่งชื่อที่ขาดมาเป็น list missing_fields เอเจนต์จึงเพิ่ม props ได้โดยไม่ต้อง parse ข้อความ component ยังต้องรับ props เป็น record เดียว: component Greeting(name: string) เป็น E0229 ที่ parameter เพราะ React ส่งแค่ props object เสมอ และ element นั้นไม่มีวันตรวจ type ผ่าน ถ้าคุณเรียกมันเป็นฟังก์ชันแทนที่จะ render ให้ประกาศด้วย fn

<match> ใน JSX ถูกตรวจเหมือน match ไหม?

มันคือสิ่งเดียวกัน แต่ละ <case> รับ pattern แบบเดียวกับที่ arm ของ match รับ และคอมไพเลอร์ส่งมันผ่านโค้ดเดียวกับที่ใช้กับ arm ชื่อที่ case bind ได้ type ของ payload field ที่ record ของ payload ไม่มีคือ E0210 และ variant ที่ขาดไปคือ E0200 case ที่ทดสอบค่าครอบคลุมแค่ส่วนหนึ่งของ variant จึงต้องมี case ข้างเคียงหรือ <case else> รับส่วนที่เหลือ:

results.glyph 
type Page =
  | Loading
  | Loaded({ users: Array<User>, total: number })
  | Failed(string)

component Results(props: { page: Page }) {
  return <match value={props.page}>
    <case Loading><p>Loading...</p></case>
    <case Loaded({ users, totl })><p>{users.length} of {totl}</p></case>
    <case Failed(message)><p>{message}</p></case>
  </match>
}

// E0210 type `{ users: Array<User>, total: number }` has no field `totl`; did you mean `total`?

หน้าเว็บเริ่มมันอย่างไร?

Glyph ไม่มี statement ระดับบนสุด หน้าเว็บจึงเรียก main ที่ export ไว้เอง แอปมีสอง module: ตัวหนึ่งเก็บ component อีกตัว mount มันบน #root ผ่าน std/dom และ createRoot ของ react-dom attribute ของ event เป็น snake case (on_change emit เป็น onChange ของ React) handler เป็น lambda และข้อความของ input คือ e.currentTarget.value

app.glyph 
module app

import react { useState }

pub component App() {
  let [name, set_name] = useState("")
  return <main>
    <input value={name} on_change={fn(e) { set_name(e.currentTarget.value) }} />
    <p>Hello, {name}</p>
  </main>
}
main.glyph 
module main

import react-dom/client { createRoot }
import std/dom
import std/console
import std/option { Some, None }
import app { App }

pub fn main() -> void {
  match dom.element_by_id("root") {
    Some(root) => createRoot(root).render(<App />),
    None => console.error("the page has no #root element"),
  }
}
index.html วางข้างสอง module 
<div id="root"></div>
<script type="module">
  import { main } from "./dist/gen/main.ts";
  main();
</script>

npx glyph build . --out dist/gen ตรวจ module รัน tsc --strict เทียบกับ declaration ของ React และเขียน TypeScript ลง dist/gen จากนั้น npx vite build --outDir dist/web อ่าน index.html ตาม import ไป และ bundle มันพร้อม React main ตัวนี้ไม่รับอะไรและไม่คืนอะไร เพราะสคริปต์เรียกมันโดยไม่มี argument ส่วน main(argv) -> number ที่ glyph init เขียนคือตัวที่ glyph run เรียกภายใต้ node ตัวอย่างเต็มทีละขั้น พร้อม package.json และบรรทัดติดตั้ง อยู่ที่ running Glyph in a browser page ในเอกสารอ้างอิงที่ glyph llms พิมพ์ออกมา CI build examples/apps/todo ด้วยวิธีเดียวกันและ render ใน Chromium แบบ headless

สถานะตอนนี้

ใช้ได้วันนี้

component, JSX, fragment (<>...</>), ชื่อ element แบบ member expression (<Ctx.Provider>), JSX prop spread ({...register()}), props ที่ถูกตรวจเทียบกับ component ก่อน tsc ไม่ว่าจะ import แบบไหน, hook ของ React เองที่ import ด้วยชื่อ (useState, useReducer, useEffect) โดย tsc ตรวจ type และ destructure คู่ค่าได้ (let [count, set_count] = useState(0)), useReducer ที่มี action เป็น tagged union แบบครบทุกกรณี, <match>/<case> ที่ถูกตรวจและแปลงเป็น match ด้วย pattern ของ arm, Context ผ่าน createContext/useContext และหน้าเว็บผ่าน std/dom (title, จุด mount, ที่อยู่), std/storage และ std/console ข้อความใน JSX อ่านแบบที่ TSX อ่าน <p>Don't stop</p> จึงเป็นข้อความธรรมดา ส่วน }, >, // และ HTML entity อย่าง &amp; ถูกปฏิเสธในข้อความ (E0012) พร้อมวิธีเขียนที่ควรใช้แทน การอ่าน member บน array ถูกตรวจ {items.foo} บน Array<string> จึงเป็น E0210 examples/apps/todo เป็นแอปเต็มบน React 18 ที่ bundle ด้วย Vite และ render ในเบราว์เซอร์

กำลังมา

ตัวตรวจของ Glyph ไม่ให้ type กับผลของ hook set_count("x") บน state ที่เป็นตัวเลขจึงเป็น error ของ tsc ใน build แทนที่จะเป็นของ Glyph ใน editor std/dom หยุดที่การอ่านหน้าเว็บและตั้ง title: React เป็นเจ้าของ DOM ที่มัน render จึงไม่มี setter ของ element หรือ event listener effect ต้องเขียนเป็น fn() -> void { ... } เพราะ lambda คืน expression สุดท้าย และ React จะเรียกมันเป็น cleanup type ของ props ที่เขียนแบบ inline ยังแสดงเป็น record ในข้อความเหล่านี้ ส่วนที่มีชื่อจะแสดงชื่อ การฝัง output ของ Glyph ในแอป Vite ที่มีอยู่ โดยที่ UI เป็น TypeScript อยู่ในคู่มือการ deploy
